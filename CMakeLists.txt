cmake_minimum_required(VERSION 3.24)
project(SwiftEdit VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
find_package(GUIForms 0.1 CONFIG REQUIRED COMPONENTS Application)
add_library(swiftedit_md4c STATIC third_party/md4c/md4c.c third_party/md4c/entity.c)
target_include_directories(swiftedit_md4c PUBLIC third_party/md4c)
add_library(notepad_document src/document.cpp src/file_windows.cpp src/session.cpp src/csv.cpp src/display.cpp src/search.cpp src/markdown.cpp)
target_include_directories(notepad_document PUBLIC src)
target_link_libraries(notepad_document PUBLIC GUIForms::Core PRIVATE swiftedit_md4c)
target_compile_definitions(notepad_document PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
if(NOT WIN32)
  message(FATAL_ERROR "This first Notepad file adapter requires Windows.")
endif()
enable_testing()
add_executable(notepad-tests tests/document_tests.cpp)
target_link_libraries(notepad-tests PRIVATE notepad_document)
add_test(NAME document COMMAND notepad-tests)
add_executable(swiftedit-session-tests tests/session_tests.cpp)
target_link_libraries(swiftedit-session-tests PRIVATE notepad_document)
add_test(NAME session COMMAND swiftedit-session-tests)
add_executable(swiftedit-csv-tests tests/csv_tests.cpp)
target_link_libraries(swiftedit-csv-tests PRIVATE notepad_document)
add_test(NAME csv COMMAND swiftedit-csv-tests)
add_executable(swiftedit-display-tests tests/display_tests.cpp)
target_link_libraries(swiftedit-display-tests PRIVATE notepad_document)
add_test(NAME display COMMAND swiftedit-display-tests)
add_executable(swiftedit-markdown-tests tests/markdown_tests.cpp)
target_link_libraries(swiftedit-markdown-tests PRIVATE notepad_document)
add_test(NAME markdown COMMAND swiftedit-markdown-tests)
add_executable(swiftedit-cli src/cli.cpp)
target_link_libraries(swiftedit-cli PRIVATE notepad_document)
find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell)
if(POWERSHELL_EXECUTABLE)
  add_test(NAME cli COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.ps1" -Executable "$<TARGET_FILE:swiftedit-cli>")
  set_tests_properties(cli PROPERTIES TIMEOUT 30)
endif()

option(NOTEPAD_BUILD_UI "Build the native editor and installed shared picker consumer" ON)
if(NOTEPAD_BUILD_UI)
  find_package(FileManagerDocumentPicker 1.0 CONFIG REQUIRED)
  add_library(notepad_ui src/editor.cpp src/windows.cpp src/csv_view.cpp src/markdown_view.cpp src/query_field.cpp)
  target_link_libraries(notepad_ui PUBLIC notepad_document GUIForms::Application FileManager::DocumentPickerView)
  add_executable(notepad WIN32 src/main.cpp)
  set_target_properties(notepad PROPERTIES OUTPUT_NAME SwiftEdit)
  target_link_libraries(notepad PRIVATE notepad_ui shell32)
  add_executable(notepad-editor-tests tests/editor_tests.cpp)
  target_link_libraries(notepad-editor-tests PRIVATE notepad_ui)
  add_test(NAME editor COMMAND notepad-editor-tests)
  add_executable(swiftedit-view-tests tests/view_tests.cpp)
  target_link_libraries(swiftedit-view-tests PRIVATE notepad_ui)
  add_test(NAME views COMMAND swiftedit-view-tests)
  add_executable(notepad-native-tests tests/native_tests.cpp)
  target_link_libraries(notepad-native-tests PRIVATE notepad_ui)
  # Opt-in because this creates self-closing owned native dialogs.
  option(NOTEPAD_NATIVE_TESTS "Run coordinated native window smoke" OFF)
  if(NOTEPAD_NATIVE_TESTS)
    add_test(NAME native COMMAND notepad-native-tests)
    set_tests_properties(native PROPERTIES TIMEOUT 30)
  endif()
  if(MSVC)
    target_link_options(notepad PRIVATE /ENTRY:mainCRTStartup)
  endif()
  add_custom_command(TARGET notepad POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:GUIForms::Application>" "$<TARGET_FILE_DIR:notepad>"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${GUIForms_FONT_DIR}" "$<TARGET_FILE_DIR:notepad>/fonts")
  install(TARGETS notepad RUNTIME DESTINATION .)
endif()
