cmake_minimum_required(VERSION 3.24)
project(SwiftEdit VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
find_package(GUIForms 0.1 CONFIG REQUIRED COMPONENTS Application)
add_library(swiftedit_md4c STATIC third_party/md4c/md4c.c third_party/md4c/entity.c)
target_include_directories(swiftedit_md4c PUBLIC third_party/md4c)
if(WIN32)
  set(SWIFTEDIT_FILE_SOURCES src/file_windows.cpp src/paged_windows.cpp)
elseif(APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(SWIFTEDIT_FILE_SOURCES src/file_posix.cpp src/paged_posix.cpp)
else()
  message(FATAL_ERROR "SwiftEdit has no file adapter for this platform.")
endif()
add_library(notepad_document src/document.cpp src/newlines.cpp ${SWIFTEDIT_FILE_SOURCES} src/session.cpp src/csv.cpp src/display.cpp src/search.cpp src/markdown.cpp src/characters.cpp src/save_review.cpp src/context.cpp src/terminal_buffer.cpp src/terminal_cells.cpp src/terminal_row.cpp src/terminal_page.cpp src/terminal_search.cpp src/terminal_query.cpp src/session_replace.cpp)
target_include_directories(notepad_document PUBLIC src)
target_sources(notepad_document PRIVATE src/terminal_wrap_view.cpp src/date_time.cpp)
target_link_libraries(notepad_document PUBLIC GUIForms::Core PRIVATE swiftedit_md4c)
target_compile_definitions(notepad_document PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
enable_testing()
add_executable(swiftedit-newline-tests tests/newline_tests.cpp)
target_link_libraries(swiftedit-newline-tests PRIVATE notepad_document)
add_test(NAME newlines COMMAND swiftedit-newline-tests)
add_executable(swiftedit-paged-tests tests/paged_tests.cpp)
target_link_libraries(swiftedit-paged-tests PRIVATE notepad_document)
add_test(NAME paged-file COMMAND swiftedit-paged-tests)
add_executable(notepad-tests tests/document_tests.cpp)
target_link_libraries(notepad-tests PRIVATE notepad_document)
add_test(NAME document COMMAND notepad-tests)
add_executable(swiftedit-session-tests tests/session_tests.cpp)
target_link_libraries(swiftedit-session-tests PRIVATE notepad_document)
add_test(NAME session COMMAND swiftedit-session-tests)
add_executable(swiftedit-csv-tests tests/csv_tests.cpp)
target_link_libraries(swiftedit-csv-tests PRIVATE notepad_document)
add_test(NAME csv COMMAND swiftedit-csv-tests)
add_executable(swiftedit-csv-clear-bench tests/csv_clear_bench.cpp)
target_link_libraries(swiftedit-csv-clear-bench PRIVATE notepad_document)
add_executable(swiftedit-display-tests tests/display_tests.cpp)
target_link_libraries(swiftedit-display-tests PRIVATE notepad_document)
add_test(NAME display COMMAND swiftedit-display-tests)
add_executable(swiftedit-markdown-tests tests/markdown_tests.cpp)
target_link_libraries(swiftedit-markdown-tests PRIVATE notepad_document)
add_test(NAME markdown COMMAND swiftedit-markdown-tests)
add_executable(swiftedit-character-tests tests/character_tests.cpp)
target_link_libraries(swiftedit-character-tests PRIVATE notepad_document)
add_test(NAME characters COMMAND swiftedit-character-tests)
add_executable(swiftedit-save-review-tests tests/save_review_tests.cpp)
target_link_libraries(swiftedit-save-review-tests PRIVATE notepad_document)
add_test(NAME save-review COMMAND swiftedit-save-review-tests)
add_executable(swiftedit-context-tests tests/context_tests.cpp)
target_link_libraries(swiftedit-context-tests PRIVATE notepad_document)
add_test(NAME context COMMAND swiftedit-context-tests)
add_executable(swiftedit-terminal-tests tests/terminal_tests.cpp)
target_link_libraries(swiftedit-terminal-tests PRIVATE notepad_document)
add_test(NAME terminal COMMAND swiftedit-terminal-tests)
add_executable(swiftedit-terminal-row-tests tests/terminal_row_tests.cpp)
target_link_libraries(swiftedit-terminal-row-tests PRIVATE notepad_document)
add_test(NAME terminal-row COMMAND swiftedit-terminal-row-tests)
add_executable(swiftedit-terminal-wrap-tests tests/terminal_wrap_tests.cpp)
target_link_libraries(swiftedit-terminal-wrap-tests PRIVATE notepad_document)
add_test(NAME terminal-wrap COMMAND swiftedit-terminal-wrap-tests)
add_executable(swiftedit-terminal-bench tests/terminal_bench.cpp)
add_executable(swiftedit-terminal-wrap-bench tests/terminal_wrap_bench.cpp)
target_link_libraries(swiftedit-terminal-wrap-bench PRIVATE notepad_document)
target_link_libraries(swiftedit-terminal-bench PRIVATE notepad_document)
# Opt-in measurement; timings are evidence, not portable pass/fail thresholds.
if(WIN32)
add_executable(swiftedit-terminal src/terminal_windows.cpp)
target_link_libraries(swiftedit-terminal PRIVATE notepad_document)
if(MINGW)
  target_link_options(swiftedit-terminal PRIVATE -municode)
endif()
add_test(NAME terminal-help COMMAND swiftedit-terminal --help)
add_executable(swiftedit-terminal-smoke src/terminal_windows.cpp)
target_compile_definitions(swiftedit-terminal-smoke PRIVATE SWIFTEDIT_TERMINAL_SMOKE)
target_link_libraries(swiftedit-terminal-smoke PRIVATE notepad_document)
if(MINGW)
  target_link_options(swiftedit-terminal-smoke PRIVATE -municode)
endif()
endif()
add_executable(swiftedit-cli src/cli.cpp)
target_link_libraries(swiftedit-cli PRIVATE notepad_document)
find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell)
if(POWERSHELL_EXECUTABLE)
  add_test(NAME cli COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.ps1" -Executable "$<TARGET_FILE:swiftedit-cli>")
  set_tests_properties(cli PROPERTIES TIMEOUT 30)
endif()

option(NOTEPAD_BUILD_UI "Build the native editor and installed shared picker consumer" ON)
if(NOTEPAD_BUILD_UI)
  find_package(FileManagerDocumentPicker 1.0 CONFIG REQUIRED)
  add_library(notepad_ui src/editor.cpp src/windows.cpp src/csv_view.cpp src/markdown_view.cpp src/query_field.cpp src/character_picker.cpp src/conflict.cpp src/user_paths.cpp)
  target_link_libraries(notepad_ui PUBLIC notepad_document GUIForms::Application FileManager::DocumentPickerView)
  if(APPLE)
    add_executable(notepad MACOSX_BUNDLE src/main.cpp)
    set_target_properties(notepad PROPERTIES
      MACOSX_BUNDLE_GUI_IDENTIFIER org.malkuth.swiftedit
      MACOSX_BUNDLE_BUNDLE_NAME SwiftEdit
      MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
      MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
    set_target_properties(notepad PROPERTIES
      BUILD_RPATH "@executable_path/../Frameworks"
      INSTALL_RPATH "@executable_path/../Frameworks"
      BUILD_WITH_INSTALL_RPATH TRUE)
  else()
    add_executable(notepad WIN32 src/main.cpp)
  endif()
  set_target_properties(notepad PROPERTIES OUTPUT_NAME SwiftEdit)
  target_link_libraries(notepad PRIVATE notepad_ui)
  if(WIN32)
    target_link_libraries(notepad PRIVATE shell32)
  endif()
  add_executable(notepad-editor-tests tests/editor_tests.cpp)
  target_link_libraries(notepad-editor-tests PRIVATE notepad_ui)
  add_test(NAME editor COMMAND notepad-editor-tests)
  add_executable(swiftedit-editor-navigation-bench tests/editor_navigation_bench.cpp)
  target_link_libraries(swiftedit-editor-navigation-bench PRIVATE notepad_ui)
  add_executable(swiftedit-csv-view-bench tests/csv_view_bench.cpp)
  target_link_libraries(swiftedit-csv-view-bench PRIVATE notepad_ui)
  add_executable(swiftedit-view-tests tests/view_tests.cpp)
  target_link_libraries(swiftedit-view-tests PRIVATE notepad_ui)
  add_test(NAME views COMMAND swiftedit-view-tests)
  add_executable(notepad-native-tests tests/native_tests.cpp)
  target_link_libraries(notepad-native-tests PRIVATE notepad_ui)
  # Opt-in because this creates self-closing owned native dialogs.
  option(NOTEPAD_NATIVE_TESTS "Run coordinated native window smoke" OFF)
  if(NOTEPAD_NATIVE_TESTS)
    add_test(NAME native COMMAND notepad-native-tests)
    set_tests_properties(native PROPERTIES TIMEOUT 30)
  endif()
  if(MSVC)
    target_link_options(notepad PRIVATE /ENTRY:mainCRTStartup)
  endif()
  if(APPLE)
    add_custom_command(TARGET notepad POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:notepad>/Contents/Frameworks"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:GUIForms::Application>"
        "$<TARGET_BUNDLE_DIR:notepad>/Contents/Frameworks/$<TARGET_FILE_NAME:GUIForms::Application>"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:GUIForms::Application>"
        "$<TARGET_BUNDLE_DIR:notepad>/Contents/Frameworks/$<TARGET_SONAME_FILE_NAME:GUIForms::Application>"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${GUIForms_FONT_DIR}"
        "$<TARGET_BUNDLE_DIR:notepad>/Contents/Resources/fonts"
      VERBATIM)
  else()
    add_custom_command(TARGET notepad POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:GUIForms::Application>" "$<TARGET_FILE_DIR:notepad>"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${GUIForms_FONT_DIR}" "$<TARGET_FILE_DIR:notepad>/fonts")
  endif()
  install(TARGETS notepad BUNDLE DESTINATION . RUNTIME DESTINATION .)
  install(FILES third_party/unicode/LICENSE.txt DESTINATION licenses RENAME Unicode.txt)
  install(FILES third_party/md4c/LICENSE.md DESTINATION licenses RENAME MD4C.md)
endif()
